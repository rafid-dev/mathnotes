% Basic document class setup
\documentclass[12pt]{book}
\usepackage{amsmath}
\usepackage{amsthm}
\usepackage{amssymb}
\usepackage[a4paper, left=0.8in, right=0.8in, top=0.8in, bottom=0.8in]{geometry}
\usepackage{tcolorbox}
\usepackage{xcolor}

% Define custom colors for boxes
\definecolor{definitionbg}{HTML}{FFFDE7}      % Light Yellow Background
\definecolor{definitiontitle}{HTML}{FBC02D}   % Dark Yellow Title Bar
\definecolor{theorembg}{HTML}{E3F2FD}         % Light Blue Background
\definecolor{theoremtitle}{HTML}{61bcff}      % Dark Blue Title Bar
\definecolor{proofcolor}{HTML}{000000}

% Definition Box Style
\newtcolorbox[auto counter, number within=section]{definitionbox}[2][]{
    colback=definitionbg,                     % Background color
    colframe=definitiontitle,                 % Border color
    coltitle=black,                           % Title text color
    fonttitle=\bfseries,                      % Title font style
    colbacktitle=definitiontitle,             % Title bar background color
    title=Definition~\thetcbcounter: #2,      % Title format
    sharp corners=south,                      % Rounded corners
    boxrule=0.75mm,                           % Border thickness
    toptitle=2mm,                             % Title bar top padding
    bottomtitle=2mm,                          % Title bar bottom padding
    #1                                        % Allow for optional arguments
}

% Theorem Box Style
\newtcolorbox[auto counter, number within=section]{theorembox}[2][]{
    colback=theorembg,                        % Background color
    colframe=theoremtitle,                    % Border color
    coltitle=black,                           % Title text color
    fonttitle=\bfseries,                      % Title font style
    colbacktitle=theoremtitle,                % Title bar background color
    title=Theorem~\thetcbcounter: #2,         % Title format
    sharp corners=south,                      % Rounded corners
    boxrule=0.75mm,                           % Border thickness
    toptitle=2mm,                             % Title bar top padding
    bottomtitle=2mm,                          % Title bar bottom padding
    #1                                        % Allow for optional arguments
}

% Define the Example Environment
\newcounter{example}[section]                 % Reset example counter at each new section
\renewcommand{\theexample}{\thesection.\arabic{example}}

\newenvironment{example}[1][]{%
    \refstepcounter{example}% Increment the counter
    \noindent\textbf{Example~\theexample}% Example heading
    \ifx\\#1\\% Check if title is provided
        : % If no title, add a colon
    \else
        \ (\textit{#1}): % If title exists, format it
    \fi
    \par\nobreak\noindent
}{%
    \par\noindent\hfill$\triangle$% End with a triangle symbol
}

% Customize the proof environment
\renewenvironment{proof}[1][Proof]{%
    \par\noindent\textit{\textbf{\textcolor{proofcolor}{#1:}}}\ }{%
    \hfill$\qedsymbol$\par
}
